\section{I. Đặt vấn đề và Ký hiệu}

\begin{frame}{1.1. Đặt vấn đề và Động lực}
  \begin{columns}[c]
    \begin{column}{0.52\textwidth}
      \begin{block}{\textbf{Kỳ vọng về hệ Multi-Agent LLM}}
        \begin{itemize}\small
          \item Phân chia vai trò: \textbf{Planner} $\to$ \textbf{Solver} $\to$ \textbf{Verifier} $\to$ \textbf{Aggregator}.
          \item Trực giác: \emph{Nhiều mô hình cùng kiểm tra chéo sẽ tốt hơn một mô hình đơn lẻ.}
          \item Thường được báo cáo tăng từ $+5\%$ đến $+15\%$.
        \end{itemize}
      \end{block}
      
      \begin{alertblock}{\textbf{Thực tế thực nghiệm bất thường}}
        \begin{itemize}\small
          \item Mô hình mạnh (7B) tự giải đúng: \textbf{46,4\%} (trên MATH).
          \item Khi kèm lời giải \textbf{sai} của mô hình yếu (1.5B): Độ chính xác của 7B tụt còn \textbf{19,2\%} (giảm $-27{,}2$ điểm!).
        \end{itemize}
      \end{alertblock}
    \end{column}
    
    \begin{column}{0.45\textwidth}
      \begin{exampleblock}{\textbf{Câu hỏi nghiên cứu cốt lõi}}
        \centering\vspace{6pt}
        \textit{Khi nào phối hợp đa tác tử thực sự tạo ra giá trị gia tăng, và khi nào nó chỉ là ảo ảnh của việc chọn sai mốc so sánh?}
        \vspace{6pt}
      \end{exampleblock}
      
      \begin{block}{\textbf{Phạm vi khảo sát}}
        \begin{itemize}\footnotesize
          \item 4 benchmarks: GSM8K, MATH-500, MBPP, HumanEval.
          \item Dải mô hình: 1,5B đến 32B (Qwen, Llama, DeepSeek).
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{1.2. Ký hiệu và Khung phân tích}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{1. Ba trạng thái mô hình}}
        \begin{itemize}\footnotesize
          \item \textbf{$W$ (Weak model):} Mô hình yếu tự giải bài toán $\to$ sinh ra \textbf{artifact} (lời giải).
          \item \textbf{$I$ (Independent strong):} Mô hình mạnh giải độc lập, không xem bài của ai.
          \item \textbf{$E$ (Exposed strong):} Cùng mô hình mạnh, nhưng ngữ cảnh kèm artifact của $W$.
        \end{itemize}
      \end{block}
      
      \begin{exampleblock}{\textbf{2. Thước đo hiệu quả}}
        \begin{itemize}\footnotesize
          \item \textbf{Hiệu quả thực tế:} $\dhonest = \acc(E) - \acc(I)$
          \item \textbf{Mức cải thiện tối đa (Trần lý tưởng):} $\dceil = G - L + R$
        \end{itemize}
      \end{exampleblock}
    \end{column}
    
    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{3. Khung 3 câu hỏi cơ chế}}
        \begin{itemize}\footnotesize
          \item \textbf{Cơ hội cải thiện ($G$):}
          $$P(W \text{ đúng} \wedge I \text{ sai})$$
          \textit{Mô hình yếu có giải đúng câu mô hình mạnh làm sai?}
          
          \item \textbf{Tổn thất tương tác ($L$):}
          $$P(W \text{ sai} \wedge I \text{ đúng} \wedge E \text{ sai})$$
          \textit{Mô hình mạnh có bị làm hỏng câu nó vốn tự làm đúng?}
          
          \item \textbf{Khả năng khai thác ($\kappa$):}
          \textit{Bộ chọn có nhặt đúng đáp án chính xác trong pool?}
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\section{II. Ba phép kiểm soát}

\begin{frame}{2.1. Sai lệch đo lường trong y văn}
  \begin{columns}[c]
    \begin{column}{0.32\textwidth}
      \begin{alertblock}{\textbf{1. Bỏ qua Compute}}
        \small
        So sánh một pipeline 4 vai (tốn $3\times - 6\times$ token) với 1 mô hình nhỏ chạy 1 lần.
        \vspace{4pt}
        
        $\implies$ \textit{Không công bằng về ngân sách tính toán (FLOPs).}
      \end{alertblock}
    \end{column}
    
    \begin{column}{0.32\textwidth}
      \begin{alertblock}{\textbf{2. Chọn sai Baseline}}
        \small
        Đo mức tăng so với mô hình yếu $W$ thay vì so với mô hình mạnh $I$.
        \vspace{4pt}
        
        $\implies$ \textit{Người dùng cần biết: Gọi thẳng $I$ hay chạy multi-agent?}
      \end{alertblock}
    \end{column}
    
    \begin{column}{0.32\textwidth}
      \begin{alertblock}{\textbf{3. Pha loãng mẫu số}}
        \small
        Tính trung bình cả các bài quá dễ hoặc quá khó mà hệ thống không thể can thiệp.
        \vspace{4pt}
        
        $\implies$ \textit{Pha loãng và che lấp hiệu ứng thực tế.}
      \end{alertblock}
    \end{column}
  \end{columns}
  
  \vspace{10pt}
  \begin{block}{\textbf{Kết luận phương pháp luận}}
    \centering\small
    Bắt buộc phải áp dụng đồng thời \textbf{3 phép kiểm soát} để đánh giá chính xác giá trị thực.
  \end{block}
\end{frame}

\begin{frame}{2.2. Ba phép kiểm soát đối chứng}
  \begin{columns}[t]
    \begin{column}{0.32\textwidth}
      \begin{block}{\textbf{1. Kiểm soát Compute}}
        \begin{itemize}\footnotesize
          \item Quy đổi FLOPs theo số tham số: Model 7B tốn $\sim 4{,}8\times$ FLOPs so với 1.5B.
          \item Pipeline sinh token gấp $2{,}9\times$ đến $6{,}6\times$.
          \item Đối chứng: So sánh ở \textbf{cùng tổng ngân sách FLOPs}.
        \end{itemize}
      \end{block}
    \end{column}
    
    \begin{column}{0.32\textwidth}
      \begin{block}{\textbf{2. Kiểm soát Baseline}}
        \begin{itemize}\footnotesize
          \item Đo đồng thời 2 mốc:
            \begin{itemize}\scriptsize
              \item $\Delta_W = \acc(\text{Hệ}) - \acc(W)$
              \item $\dhonest = \acc(\text{Hệ}) - \acc(I)$
            \end{itemize}
          \item Giá trị thực: Chỉ khi $\dhonest > 0$ thì hệ thống mới có ích.
        \end{itemize}
      \end{block}
    \end{column}
    
    \begin{column}{0.32\textwidth}
      \begin{block}{\textbf{3. Kiểm soát Mẫu số}}
        \begin{itemize}\footnotesize
          \item Phân tầng tập MATH:
            \begin{itemize}\scriptsize
              \item $32\%$ bài: Luôn sai.
              \item $25\%$ bài: Luôn đúng.
            \end{itemize}
          \item $57\%$ bài nằm ngoài khả năng can thiệp $\implies$ Phân tích trên tập bài có biến thiên.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{3.2. Cơ chế: Truyền dẫn vs Sửa lỗi}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Bản chất giao thức Sửa bài (Correction)}}
        \begin{itemize}\small
          \item Model sau đọc chuỗi suy luận của model trước.
          \item \textbf{Phát hiện:} Model đóng vai trò \emph{kênh truyền dẫn đáp án} hơn là tự kiểm tra và sửa lỗi logic.
          \item Nếu artifact ban đầu sai, mô hình mạnh bị định hướng sai và lặp lại lỗi.
        \end{itemize}
      \end{block}
    \end{column}
    
    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{\textbf{Phân loại hai lớp giao thức}}
        \begin{itemize}\small
          \item \textbf{1. Tuyển chọn (Selection):}
            \begin{itemize}\footnotesize
              \item Chỉ nhặt lời giải tốt nhất từ pool độc lập.
              \item Đặc tính: $L = 0$ \textbf{theo cấu trúc} (không làm hỏng bài đúng).
            \end{itemize}
          \item \textbf{2. Sửa chữa (Correction / Repair):}
            \begin{itemize}\footnotesize
              \item Sinh chuỗi mới dựa trên ngữ cảnh cũ.
              \item Đặc tính: $L > 0$ (nguy cơ tổn thất cao do phơi nhiễm nhiễu).
            \end{itemize}
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{4.1. Đóng góp từng vai trò}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Đóng góp từng vai trò ($P, S, V, A$)}}
        \begin{itemize}\small
          \item \textbf{Solver ($S$):} Đóng góp $>90\%$ giá trị hệ thống. Năng lực của Solver quyết định trần hiệu năng.
          \item \textbf{Planner ($P$):} Đóng góp $\approx 0$ trên các tác vụ toán và code có cấu trúc.
        \end{itemize}
      \end{block}
    \end{column}
    
    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Bộ kiểm ($V$) \& Bộ tổng hợp ($A$)}}
        \begin{itemize}\small
          \item \textbf{Verifier ($V$):} Đóng góp phụ thuộc chặt chẽ vào độ chính xác của tín hiệu kiểm chứng.
          \item \textbf{Aggregator ($A$):} Tổng hợp bằng LLM không vượt trội hơn Bỏ phiếu đa số cơ học.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
  
  \vspace{8pt}
  \begin{alertblock}{\textbf{Đúc kết}}
    \centering\small
    Trong hệ 4 vai trò, Solver là trụ cột duy nhất gánh vác toàn bộ hiệu năng; các vai trò còn lại đóng góp rất khiêm tốn hoặc trung tính.
  \end{alertblock}
\end{frame}

\begin{frame}{4.2. Năng lực Bộ kiểm (Verifier)}
  \begin{columns}[t]
    \begin{column}{0.50\textwidth}
      \begin{block}{\textbf{Kiểm chứng tất định (Code Execution)}}
        \centering\footnotesize
        \begin{tabular}{lcccc}\toprule
          \textbf{Cấu hình} & \textbf{Greedy} & \textbf{\texttt{exec3}} & \textbf{Phá bài}\\\midrule
          HE 1.5B (Kaggle) & 0,5375 & \textbf{0,6000} & \textbf{0,0}\\
          HE 1.5B (5090)   & 0,5625 & \textbf{0,6438} & \textbf{0,0}\\
          HE 7B (5090)     & 0,8000 & \textbf{0,8812} & \textbf{0,0}\\

        \end{tabular}
        \vspace{4pt}
        \begin{itemize}\scriptsize
          \item Đạt đúng trần \textbf{Oracle@4}, phá $0$ bài trong 20/20 fold.

        \end{itemize}
      \end{block}
    \end{column}
    
    \begin{column}{0.46\textwidth}

          \item Hiệu năng tụt dưới mức greedy baseline.
        \end{itemize}

        \begin{itemize}\small

          \item $\implies$ Verifier không hoàn hảo sẽ làm hại khi Solver đã mạnh.
        \end{itemize}
      \end{block}

          \item Ngưỡng đổi dấu $g^*$ cần thêm khoảng tin cậy tham số mở rộng.

      \end{block}
      
      \begin{block}{\textbf{4. Gán công trạng}}
        \begin{itemize}\small
          \item Phân tích Shapley dựa trên bootstrap theo mẫu bài toán mà chưa theo fold dữ liệu hoàn toàn độc lập.
